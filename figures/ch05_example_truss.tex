  \draw[dim] (0,-0.6) -- (3,-0.6) node[midway,fill=white,inner sep=1pt] {3\,m};
  \draw (0,-0.8) -- (0,-0.4); \draw (3,-0.8) -- (3,-0.4);
  % axes
  \draw[->] (-0.9,-1.1) -- (-0.3,-1.1) node[right] {$x$};
